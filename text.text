To Create own Artisan command to automatically create:-
    * (Name) Controller
    * Model
    * migration
    * (optional) routes/views
       Commend:-
            php artisan make:command NameOfTheCommand
       Code :-
            <?php

                namespace App\Console\Commands;

                use Illuminate\Console\Command;
                use Illuminate\Support\Str;
                use Illuminate\Support\Facades\Artisan;

                class MakeAdminCrud extends Command
                {
                    protected $signature = 'make:admin {name}';
                    protected $description = 'Create Admin Model, Controller, Migration';

                    public function handle()
                    {
                        $name = Str::studly($this->argument('name'));

                        // Create Model + Migration
                        Artisan::call("make:model {$name} -m");

                        // Create Admin Controller
                        Artisan::call("make:controller Admin/{$name}Controller");

                        $this->info("Admin CRUD created for {$name}");
                    }
                }
            ?>

admincontroller.php
    public function auth(Request $request)
    {
        //this is for admin authentication
        $email = $request->post('email');
        $password = $request->post('password');

        $result = Admin::where(['email' => $email, 'password' => $password])->first();
        if ($result) {
            $request->session()->put('ADMIN_LOGIN', true);
            $request->session()->put('ADMIN_ID', $request->id);
            return redirect('admin/dashboard')->with('success', 'You are logged in successfully');
        } else {
            return redirect('admin')->with('error', 'Please enter valid details');
        }
    }
