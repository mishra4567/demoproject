    public function manageproductprocess(Request $request)
    {
        // Code frome chatgpt
        $id = $request->post('id');

        if($request->post('id')>0){
            $image_validation = 'mimes:png,jpg,jpeg';
        }else{
            $image_validation = 'required|mimes:png,jpg,jp,webp';
        }

        $request->validate([
            'name' => 'required',
            'image' => $image_validation,
            'slug' => [
                'required',
                Rule::unique('products', 'slug')->ignore($id),
            ],
        ], [
            'slug.unique' => 'This Product already exists',
        ]);

        $model = $id ? Product::findOrFail($id) : new Product();
        if ($request->hasFile('image')) {
            $image = $request->file('image');
            $ext = $image->extension();
            $image_name = time() . '.' . $ext;
            $image->move(public_path('/storage/media'), $image_name);
            $model->image = $image_name;
        }

        $model->category_id = $request->category_id;
        $model->name = $request->name;
        $model->slug = $request->slug;
        $model->brand = $request->brand;
        $model->model = $request->model;
        $model->short_desc = $request->short_desc;
        $model->desc = $request->desc;
        $model->keywords = $request->keywords;
        $model->technical_specification = $request->technical_specification;
        $model->uses = $request->uses;
        $model->warranty = $request->warranty;
        $model->status = 1;
        $model->save();

        return redirect('admin/product')
            ->with('success', $id ? 'Product Updated Successfully' : 'Product Inserted Successfully');


        // // return $request->post();
    }

    public function manageproduct(Request $request, $id = '')
